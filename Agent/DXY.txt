export REDISCLI_AUTH='xau12345'

DEV='dev_970a8ac0e6c745259e5b820a2e4c5d8a'

LAST_DIR=""
LAST_CONF=""
LAST_BULL_BREAK=""
LAST_BEAR_BREAK=""
LAST_KEY=""

while true; do
    K=$(
        redis-cli --scan \
          --pattern "xtl:dxy:features:H1:REAL_DXY:${DEV}:*" |
        awk -F: '
          BEGIN {
            max = 0
            key = ""
          }
          {
            ts = $NF + 0
            if (ts > max) {
              max = ts
              key = $0
            }
          }
          END {
            print key
          }
        '
    )

    if [ -z "$K" ]; then
        echo "$(date -u) no DXY H1 feature key found"
        sleep 30
        continue
    fi

    RAW=$(redis-cli --raw GET "$K" 2>/dev/null)

    if ! printf '%s\n' "$RAW" | jq -e . >/dev/null 2>&1; then
        echo "$(date -u) invalid JSON key=$K"
        sleep 30
        continue
    fi

    DIR=$(printf '%s\n' "$RAW" |
        jq -r '.candidate_direction // "UNKNOWN"')

    CONF=$(printf '%s\n' "$RAW" |
        jq -r '.candidate_confidence // 0')

    BULL_BREAK=$(printf '%s\n' "$RAW" |
        jq -r '.bullish_structure_break // false')

    BEAR_BREAK=$(printf '%s\n' "$RAW" |
        jq -r '.bearish_structure_break // false')

    CHANGED=false

    [ "$DIR" != "$LAST_DIR" ] && CHANGED=true
    [ "$CONF" != "$LAST_CONF" ] && CHANGED=true
    [ "$BULL_BREAK" != "$LAST_BULL_BREAK" ] && CHANGED=true
    [ "$BEAR_BREAK" != "$LAST_BEAR_BREAK" ] && CHANGED=true

    # Also print once for every newly published M15 snapshot.
    if [ "$K" != "$LAST_KEY" ]; then
        NEW_BAR=true
    else
        NEW_BAR=false
    fi

    if [ "$CHANGED" = true ] || [ "$NEW_BAR" = true ]; then
        echo
        echo "=================================================="
        echo "UTC time: $(date -u)"
        echo "Key: $K"
        echo
        echo "Previous:"
        echo "  direction=$LAST_DIR"
        echo "  confidence=$LAST_CONF"
        echo "  bullish_break=$LAST_BULL_BREAK"
        echo "  bearish_break=$LAST_BEAR_BREAK"
        echo
        echo "Current:"

        printf '%s\n' "$RAW" |
        jq '{
          source,
          device_id,
          broker_bar_open_ms,
          broker_bar_close_ms,
          bar_close_ms,
          detected_at_ms,

          candidate_direction,
          candidate_confidence,

          bullish_structure_break,
          bearish_structure_break,

          recent_net_atr,
          prior_net_atr,
          slope_5_atr,
          prior_slope_5_atr,
          slope_12_atr,
          acceleration_atr,

          up_steps,
          down_steps,
          bullish_bodies,
          bearish_bodies,

          pin_bar_direction,
          bullish_pin_score,
          bearish_pin_score,

          compression_ratio,
          expansion_ratio,
          body_ratio,

          nearest_H1_support:
            .sr_audit.H1.nearest_support,

          nearest_H1_resistance:
            .sr_audit.H1.nearest_resistance
        }'

        if [ "$BULL_BREAK" = true ] &&
           [ "$LAST_BULL_BREAK" != true ]; then
            echo
            echo ">>> NEW BULLISH STRUCTURE BREAK"
        fi

        if [ "$BEAR_BREAK" = true ] &&
           [ "$LAST_BEAR_BREAK" != true ]; then
            echo
            echo ">>> NEW BEARISH STRUCTURE BREAK"
        fi

        if [ "$DIR" != "$LAST_DIR" ]; then
            echo
            echo ">>> CLASSIFIER CHANGE: $LAST_DIR -> $DIR"
        fi

        if [ "$NEW_BAR" = true ]; then
            echo
            echo ">>> NEW H1 SNAPSHOT"
        fi
    fi

    LAST_DIR="$DIR"
    LAST_CONF="$CONF"
    LAST_BULL_BREAK="$BULL_BREAK"
    LAST_BEAR_BREAK="$BEAR_BREAK"
    LAST_KEY="$K"

    sleep 30
done

from datetime import datetime, timezone

from api.strategy.oppt_executor import R

)

    summarize_sr_multi_tf,
    build_market_structure_state,
)

H1_MIN_SCORE = 40
H1_MIN_MARGIN = 20

def dec(x):
    if isinstance(x,(bytes,bytearray)):
        x=x.decode("utf-8","replace")
    try:
        return json.loads(x or "{}")
    except:
        return {}

def ts(ms):
    return datetime.fromtimestamp(
        int(ms)/1000,
        tz=timezone.utc,
    ).strftime("%Y-%m-%d %H:%M:%S UTC") if ms else None

c=dec(R.get("xtl:dxy:canonical"))
dev=c.get("device_id") or c.get("real_device_id")

m=read_dxy_m15_state(R,"REAL_DXY",dev) or {}
f=m.get("features") or {}

    f.get("bar_close_ms")
    or f.get("broker_bar_close_ms")
    or m.get("last_evaluated_bar_close_ms")
    or 0
)

old=((f.get("market_flow") or {}).get("structure") or {})
px=float(old.get("price") or 0)

h1f=dec(R.get(
    f"xtl:dxy:features:latest:H1:REAL_DXY:{dev}"
))

bull=int(h1f.get("bull_evidence_score") or 0)
bear=int(h1f.get("bear_evidence_score") or 0)
margin=abs(bull-bear)

if bull >= H1_MIN_SCORE and bull-bear >= H1_MIN_MARGIN:
    h1_dir="BULLISH"
elif bear >= H1_MIN_SCORE and bear-bull >= H1_MIN_MARGIN:
    h1_dir="BEARISH"
else:
    h1_dir="NEUTRAL"

m15_dir=str(m.get("direction") or "NEUTRAL").upper()

if h1_dir in ("BULLISH","BEARISH") and m15_dir == h1_dir:
    combined="AGREE"
    combined_dir=h1_dir
elif h1_dir in ("BULLISH","BEARISH") and m15_dir in ("BULLISH","BEARISH"):
    combined="CONFLICT"
    combined_dir="NEUTRAL"
else:
    combined="DEVELOPING"
    combined_dir="NEUTRAL"

h1=_load_native_real_dxy_htf_bars(
    device_id=dev,tf="H1",cutoff_ms=cutoff,
